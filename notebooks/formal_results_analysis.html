Formal Experiment Analysis — L1 (Local Index) vs S1 (Shared Index)

Analysis of the 18-run formal study (2 architectures x 3 workloads x 3 repetitions) defined in experiment-config.json and produced by scripts/formal/. All 18 runs are valid (see results/formal/<system>/<workload>/r<n>/validity.json).

  • L1 (local_index): each node has local disk-attached storage.
  • S1 (shared_index): compute/storage disaggregation via OpenSearch remote-backed storage (MinIO/S3).
  • W0: read-only baseline (50 QPS, no writes).
  • W1: 50 QPS + 250 docs/s concurrent indexing.
  • W2: 50 QPS + 500 docs/s concurrent indexing (heavy).

This notebook deliberately separates two distinct questions that are easy to conflate:

1. Indexing-induced degradation: does concurrent indexing raise query latency relative to each architecture's own read-only baseline (W0)? This is the primary research question and is answered by the degradation ratio D = latency(W1 or W2) / latency(W0), computed per architecture, per repetition — never by comparing L1 to S1 directly.

2. Absolute latency: which architecture is faster in absolute terms at a given workload? This is a secondary, descriptive comparison.

With only n=3 repetitions per cell, standard deviation is unstable and can visually overstate confidence. The primary figures below therefore show individual observations, the median, and the min-max range rather than mean +/- std bars. Mean +/- std is kept further down for reference only.

def _load_json(path: Path) -> dict | None:
    if not path.exists():
        return None
    with open(path) as f:
        return json.load(f)


def _container_totals(container_summary: dict | None) -> dict:
    if not container_summary:
        return {"cpu_pct_avg_total": np.nan, "cpu_pct_max_total": np.nan, "mem_mb_avg_total": np.nan, "mem_mb_max_total": np.nan}
    return {
        "cpu_pct_avg_total": sum(c["cpu_pct_avg"] for c in container_summary.values()),
        "cpu_pct_max_total": max(c["cpu_pct_max"] for c in container_summary.values()),
        "mem_mb_avg_total": sum(c["mem_mb_avg"] for c in container_summary.values()),
        "mem_mb_max_total": max(c["mem_mb_max"] for c in container_summary.values()),
    }


def _query_p99_ms(run_dir: Path, lead_in_s: float) -> float:
    # latency_ms_p99_overall is not written to query_summary.json, so p99 is
    # computed here from the same raw per-request log the summary itself is
    # derived from, filtered to the same measurement window (tick_offset_s >= lead_in_s).
    path = run_dir / "queries.csv"
    if not path.exists():
        return np.nan
    df = pd.read_csv(path)
    df = df[df["tick_offset_s"] >= lead_in_s]
    return float(df["latency_ms"].quantile(0.99))


def _segment_merge_delta(before: dict | None, after: dict | None) -> dict:
    # "primaries" = primary shards only; "total" = primaries + replicas (cluster-wide
    # merge work). L1 runs with a replica that has its own independent merge activity,
    # so the "total" figures are the ones that matter for total merge/CPU cost —
    # the "primaries" figures are kept alongside, explicitly labeled, for reference.
    if before is None or after is None:
        return {
            "segments_before": np.nan, "segments_after": np.nan,
            "segments_before_total": np.nan, "segments_after_total": np.nan,
            "merges_total_delta_primaries": np.nan, "merges_time_ms_delta_primaries": np.nan,
            "merges_total_delta_total": np.nan, "merges_time_ms_delta_total": np.nan,
        }
    bp, ap = before["_all"]["primaries"], after["_all"]["primaries"]
    bt, at = before["_all"]["total"], after["_all"]["total"]
    return {
        "segments_before": bp["segments"]["count"],
        "segments_after": ap["segments"]["count"],
        "segments_before_total": bt["segments"]["count"],
        "segments_after_total": at["segments"]["count"],
        "merges_total_delta_primaries": ap["merges"]["total"] - bp["merges"]["total"],
        "merges_time_ms_delta_primaries": ap["merges"]["total_time_in_millis"] - bp["merges"]["total_time_in_millis"],
        "merges_total_delta_total": at["merges"]["total"] - bt["merges"]["total"],
        "merges_time_ms_delta_total": at["merges"]["total_time_in_millis"] - bt["merges"]["total_time_in_millis"],
    }


def _remote_transfer_bytes(before: dict | None, after: dict | None) -> dict:
    if before is None or after is None:
        return {"remote_upload_bytes_delta": np.nan, "remote_download_bytes_delta": np.nan}

    def totals(payload: dict) -> tuple[int, int]:
        up, down = 0, 0
        for node in payload["nodes"].values():
            rs = node.get("indices", {}).get("segments", {}).get("remote_store", {})
            up += rs.get("upload", {}).get("total_upload_size", {}).get("succeeded_bytes", 0)
            down += rs.get("download", {}).get("total_download_size", {}).get("succeeded_bytes", 0)
        return up, down

    up_before, down_before = totals(before)
    up_after, down_after = totals(after)
    return {"remote_upload_bytes_delta": up_after - up_before, "remote_download_bytes_delta": down_after - down_before}

Reusable dot + median + min-max plot helper

This is the primary presentation style used below: every one of the 3 repetitions is plotted as its own point, with a median marker and a min-max range bar, rather than collapsing to mean +/- std.

def plot_dot_median(ax, df, metric, title=None, ylabel=None, ref_line=None,
                    systems=("L1", "S1"), workloads=WORKLOADS_UPPER, jitter=0.04):
    x = np.arange(len(workloads))
    n = len(systems)
    span = 0.3
    offsets = {s: (i - (n - 1) / 2) * (span / max(n - 1, 1)) for i, s in enumerate(systems)}

    for system in systems:
        for i, wl in enumerate(workloads):
            vals = df[(df["system"] == system) & (df["workload"] == wl)][metric].dropna().to_numpy()
            if len(vals) == 0:
                continue
            xs = x[i] + offsets[system] + RNG.uniform(-jitter, jitter, size=len(vals))
            ax.scatter(xs, vals, color=SYSTEM_COLOR[system], alpha=0.85, zorder=3, s=45,
                       label=system if i == 0 else None)
            med = np.median(vals)
            ax.plot([x[i] + offsets[system] - 0.09, x[i] + offsets[system] + 0.09], [med, med],
                    color=SYSTEM_COLOR[system], linewidth=2.8, zorder=4)
            if len(vals) > 1:
                ax.vlines(x[i] + offsets[system], vals.min(), vals.max(),
                          color=SYSTEM_COLOR[system], alpha=0.35, linewidth=1.5, zorder=2)

    if ref_line is not None:
        ax.axhline(ref_line, color="gray", linestyle="--", linewidth=1, zorder=1)
    ax.set_xticks(x)
    ax.set_xticklabels(workloads)
    if title:
        ax.set_title(title)
    if ylabel:
        ax.set_ylabel(ylabel)
    ax.legend()

Primary result 2: p95 degradation ratio — THE key figure

D_p95 = p95(W1 or W2) / p95(W0), computed per repetition, per architecture (matching W1/W2 rep r against W0 rep r of the same architecture). This directly operationalizes the research question:

  • D > 1: concurrent indexing degraded query latency relative to that architecture's own read-only baseline.
  • D = 1: unchanged.
  • D < 1: measured latency was lower than the read-only baseline (noise, or a genuine confound — see the diagnostics below).

Sensitivity analysis: does W1/W2's 30s discarded lead-in explain their lower latency vs. W0?

This does not replace the primary result above. The reported D_p95 is the one in "Primary result 2"; this section only checks a specific procedural asymmetry: W1/W2's measured window is preceded by 30s of discarded live query+write traffic, so the system has already been running under load for 30s before their p95 is computed. W0 has no such lead-in — its p95 is computed from the very first measured second.

To check whether that head start explains W1/W2's lower latency, W0's own p95 is recomputed after dropping its first 30 measured seconds: p95(W0, 30-300s), a 270s window, using the same raw queries.csv files but a window trim that is local to this section only (the global LEAD_IN_S["w0"] stays 0.0 everywhere else in this notebook). The adjusted ratio keeps the original D = p95(load) / p95(baseline) convention — load (W1/W2, unchanged 0-300s post-lead-in window) over the adjusted baseline — so < 1 still means "load looks faster than baseline," exactly as in the primary result, just with a fairer W0 baseline.

  • If D_p95^adjusted stays materially below 1 for L1 and near 1 for S1 (essentially unchanged from the primary D_p95), the 30s lead-in asymmetry does not explain the pattern and the concern can be dismissed.
  • If it moves substantially toward or above 1, the asymmetry is a real procedural confound and should be disclosed as a limitation.
def _p95_w0_trimmed(system_dir: str, rep: int, trim_start_s: float = 30.0, trim_end_s: float = 300.0) -> float:
    # W0 has LEAD_IN_S["w0"] == 0.0 elsewhere in this notebook (untouched); this
    # trim is local to this sensitivity-analysis section only.
    path = RESULTS_ROOT / system_dir / "w0" / f"r{rep}" / "queries.csv"
    df = pd.read_csv(path)
    df = df[(df["tick_offset_s"] >= trim_start_s) & (df["tick_offset_s"] < trim_end_s)]
    return float(df["latency_ms"].quantile(0.95))

Diagnostic 5: within-run latency drift over the 300s measurement window

Buckets each run's queries into 10s windows and plots bucket-p95 over time. Reveals whether an aggregate p95 is hiding a warm-up tail, a spike, or slow drift within the window — for example, whether L1 W0 starts hot and cools, which the run-level aggregate would mask.

BUCKET_S = 10


def load_measurement_queries(system_dir: str, workload: str, rep: int) -> pd.DataFrame:
    path = RESULTS_ROOT / system_dir / workload / f"r{rep}" / "queries.csv"
    df = pd.read_csv(path)
    lead_in = LEAD_IN_S[workload]
    df = df[df["tick_offset_s"] >= lead_in].copy()
    df["time_s"] = df["tick_offset_s"] - lead_in
    return df


def bucket_p95(df: pd.DataFrame, bucket_s: float = BUCKET_S, measurement_s: float = MEASUREMENT_S) -> pd.Series:
    bins = np.arange(0, measurement_s + bucket_s, bucket_s)
    labels = bins[:-1] + bucket_s / 2
    df = df.copy()
    df["bucket_mid"] = pd.cut(df["time_s"], bins=bins, right=False, labels=labels)
    return df.groupby("bucket_mid", observed=True)["latency_ms"].quantile(0.95)


SYSTEM_DIRS = {"L1": "local_index", "S1": "shared_index"}
